cmake_minimum_required(VERSION 3.22)
project(bendwhisper C CXX)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
# The Pixel 6a's Tensor G1 (A55/A76/X1) has dotprod + fp16; q8_0 is fastest there (asr/README.md has the numbers).
set(GGML_CPU_ARM_ARCH "armv8.2-a+dotprod+fp16" CACHE STRING "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
add_subdirectory(${WHISPER_SRC} whisper)
add_library(bendwhisper SHARED whisper_jni.c)
target_link_libraries(bendwhisper whisper log)
target_link_options(bendwhisper PRIVATE -Wl,--gc-sections -Wl,-z,max-page-size=16384)
